        public sealed class Failure
        {
            public string Check { get; set; }
            public int Occurrences { get; set; }
            public string FirstCase { get; set; }
            public string Detail { get; set; }
        }
        public sealed class Control

            public string Name { get; set; }

        public sealed class Report
        {
            public int DistinctSeeds { get; set; }
            public int Cases { get; set; }
            public int ConstructedCases { get; set; }
            public int DeterministicPairs { get; set; }
            public long CellComparisons { get; set; }
            public long Assertions { get; set; }
            public int MutationCases { get; set; }
            public int RejectedMutations { get; set; }
            public int InvalidInputCases { get; set; }
            public int RejectedInputs { get; set; }
            public double Seconds { get; set; }
            public bool Success => Failures.Count == 0;
            public List<Failure> Failures { get; } = new();
            public List<Control> Controls { get; } = new();
            public Dictionary<string, int[]> CountRanges { get; } = new();
            public void AddFailure(string check, string context, string detail)
            {
                Failure f = Failures.FirstOrDefault(v => v.Check == check);
                if (f == null) Failures.Add(new() { Check = check, Occurrences = 1, FirstCase = context, Detail = detail });
                else f.Occurrences++;
            }
            internal void Check(bool ok, string code, string context, string detail)
            {
                Assertions++;
                if (!ok) AddFailure(code, context, detail);
            }
            internal void Range(string name, int value)
            {
                if (!CountRanges.TryGetValue(name, out int[] range)) CountRanges[name] = new[] { value, value };
                else { range[0] = Math.Min(range[0], value); range[1] = Math.Max(range[1], value); }
            }
        }
        private const int W = 280, H = 256;
        private static bool In(int x, int y) => x >= 4 && y >= 4 && x < W - 4 && y < H - 4;
        private static bool In(P.Point p) => In(p.X, p.Y);
        private static bool Solid(P.Cell c) => c.Write && c.Key != null && c.Slope == 0;
        private static readonly P.Point[] Directions = { new(-1, 0), new(1, 0), new(0, -1), new(0, 1) };
        private static readonly string[] RouteNames = { "left-edge", "right-edge", "rising", "falling", "sine", "bend" };
        private static readonly (int left, int right)[] Shoulders = { (40, 40), (76, 76), (40, 76), (76, 40), (48, 65), (65, 48) };

        private static int[] Seeds(int count)
        {
            var seeds = new List<int> { 0, 1, -1, 2, -2, 42, 12345, int.MinValue, int.MaxValue, unchecked((int)0x9e3779b9u) };
            uint state = 0x51ED270Bu;
            while (seeds.Count < count)
            {
                state ^= state << 13; state ^= state >> 17; state ^= state << 5;
                int value = unchecked((int)state);
                if (!seeds.Contains(value)) seeds.Add(value);
            }
            return seeds.ToArray();
        }
        private static int[] Spine(int profile)
        {
            var result = new int[H];
            for (int y = 0; y < H; y++)
                result[y] = profile switch {
                    0 => 110, 1 => 165,
                    2 => 110 + 55 * y / (H - 1), 3 => 165 - 55 * y / (H - 1),
                    4 => (int)Math.Round(137.5 + 27.5 * Math.Sin(y * Math.PI * 2 / (H - 1))),
                    _ => (int)Math.Round(137.5 - 27.5 * Math.Cos(y * Math.PI * 2 / (H - 1)))
                };
            return result;
        }
        public static Report Run(int seedCount)
        {
            if (seedCount < 128) throw new ArgumentOutOfRangeException(nameof(seedCount));
            var report = new Report(); var timer = Stopwatch.StartNew();
            report.Check(P.Width == W && P.Height == H && P.SurfaceY == 56, "contract", "dimensions", "Expected Width=280, Height=256, SurfaceY=56.");
            if (!report.Success) return report;
            int[] seeds = Seeds(seedCount); report.DistinctSeeds = seeds.Length;
            var digests = new HashSet<string>();
            for (int i = 0; i < seeds.Length; i++)
            {
                Case(report, seeds[i], null, 56, 56, "default", digests);
                int profile = i % RouteNames.Length;
                var shoulder = Shoulders[(i / RouteNames.Length) % Shoulders.Length];
                Case(report, seeds[i], Spine(profile), shoulder.left, shoulder.right, RouteNames[profile], null);
            }
            // Full route/shoulder cross-product on a fixed seed separates profile
            // coverage from seed randomness, including both legal shoulder extremes.
            for (int route = 0; route < RouteNames.Length; route++)
                foreach (var shoulder in Shoulders)
                    Case(report, 42, Spine(route), shoulder.left, shoulder.right, RouteNames[route] + "/matrix", null);
            report.Check(digests.Count > 1, "seed-variation", "default", "Distinct seeds produced no distinct successful plans.");
            InvalidInputs(report);
            Mutations(report);
            timer.Stop(); report.Seconds = timer.Elapsed.TotalSeconds;
            return report;
        }
        private static void Case(Report r, int seed, int[] spine, int left, int right, string profile, HashSet<string> digests)
        {
            string context = $"seed={seed}, route={profile}, shoulders={left}/{right}";
            r.Cases++;
            if (spine != null)
                r.Check(spine.Length == H && spine.All(x => x >= 110 && x <= 165) &&
                    spine.Zip(spine.Skip(1), (a, b) => Math.Abs(a - b)).All(d => d <= 1),
                    "harness-input", context, "Expected a smooth 256-row spine within 110..165, adjacent delta <= 1.");
            try
            {
                int[] before = spine == null ? null : (int[])spine.Clone();
                P p = P.Create(seed, spine, left, right);
                r.ConstructedCases++;
                r.Check(spine == null || spine.SequenceEqual(before), "input-mutated", context, "Create modified its caller-owned spine array.");
                // A separate array allocation must yield the same complete public plan.
                P q = P.Create(seed, before == null ? null : (int[])before.Clone(), left, right);
                string digest = Fingerprint(p);
                bool same = digest == Fingerprint(q);
                r.Check(same, "determinism-metadata", context, "Complete cells, arrays, ordered objects or rib roots differ.");
                bool cellsEqual = true;
                for (int y = 0; y < H; y++) for (int x = 0; x < W; x++)
                { cellsEqual &= p.Cells[x, y] == q.Cells[x, y]; r.CellComparisons++; }
                r.Check(cellsEqual, "determinism-cells", context, "Repeated seed/input changed cells.");
                if (same && cellsEqual) r.DeterministicPairs++;
                digests?.Add(digest);
                CheckPlan(p, r, context);
                r.Check(Fingerprint(p) == digest, "query-mutated-plan", context, "Validation/reachability changed plan data.");
            }
            catch (Exception e) { r.AddFailure("case-exception/" + e.GetType().Name + "/" + e.Message, context, e.Message); }
        }
        private static string Fingerprint(P p)
        {
            using var memory = new MemoryStream();
            using var writer = new BinaryWriter(memory, Encoding.UTF8, true);
            void Text(string value) { writer.Write(value != null); if (value != null) writer.Write(value); }
            void Point(P.Point at) { writer.Write(at.X); writer.Write(at.Y); }
            writer.Write(P.Version); writer.Write(W); writer.Write(H); Point(p.Entrance); Point(p.Exit);
            for (int y = 0; y < H; y++) for (int x = 0; x < W; x++)
            { var c = p.Cells[x, y]; Text(c.Key); Text(c.Wall); writer.Write(c.Slope); writer.Write(c.Side); writer.Write(c.Write); }
            writer.Write(p.Ground.Length); foreach (int v in p.Ground) writer.Write(v);
            writer.Write(p.Center.Length); foreach (int v in p.Center) writer.Write(v);
            writer.Write(p.Teeth.Count); foreach (var t in p.Teeth) { Point(t.Root); writer.Write(t.Variant); }
            writer.Write(p.Fibers.Count); foreach (var f in p.Fibers) { Point(f.Root); writer.Write(f.Length); }
            writer.Write(p.Ribs.Count); foreach (var rib in p.Ribs)
            { Point(rib.Root); Point(rib.Tip); writer.Write(rib.Roots.Length); foreach (var at in rib.Roots) Point(at); }
            writer.Write(p.Pockets.Count); foreach (var pocket in p.Pockets)
            { Point(pocket.Center); writer.Write(pocket.RadiusX); writer.Write(pocket.RadiusY); Text(pocket.Role); }
            writer.Write(p.Chests.Count); foreach (var at in p.Chests) Point(at);
            writer.Write(p.NodeSites.Count); foreach (var at in p.NodeSites) Point(at);
            writer.Flush(); return Convert.ToHexString(SHA256.HashData(memory.ToArray()));
        }
        private static void CheckPlan(P p, Report r, string context)
        {
            void Check(bool ok, string code, string detail) => r.Check(ok, code, context, detail);
            p.Validate();
            Check(p.Cells.GetLength(0) == W && p.Cells.GetLength(1) == H && p.Center.Length == H && p.Ground.Length == W,
                "array-shapes", "Wrong cell/center/ground dimensions.");
            int active = 0, sloped = 0, amber = 0;
            var materials = new HashSet<string> { "cap", "grass", "soil", "stone", "rib", "amber" };
            var walls = new HashSet<string> { "soil", "grass", "stone", "amber" };
            for (int y = 0; y < H; y++) for (int x = 0; x < W; x++)
            {
                var c = p.Cells[x, y];
                // Avoid allocating millions of success messages during seed sweeps.
                void CellCheck(bool ok, string code)
                {
                    r.Assertions++;
                    if (!ok) r.AddFailure(code, context, $"Cell ({x},{y}): key={c.Key}, wall={c.Wall}, slope={c.Slope}, side={c.Side}, write={c.Write}.");
                }
                CellCheck(In(x, y) || c == default, "padding");
                CellCheck(c.Write || c == default, "inactive-cell");
                CellCheck(c.Key == null || materials.Contains(c.Key), "material");
                CellCheck(c.Wall == null || walls.Contains(c.Wall), "wall-material");
                CellCheck(c.Slope <= 4 && (c.Slope == 0 || c.Key == "rib"), "native-slope");
                CellCheck((c.Key != "amber" && c.Wall != "amber") || c.Side, "side-amber");
                if (In(x, y)) CellCheck(y >= p.Ground[x] + 3 || c.Wall == null, "wall-fringe");
                if (c.Write) active++; if (c.Slope != 0) sloped++; if (c.Key == "amber" || c.Wall == "amber") amber++;
            }
            Check(active > 0 && active < (W - 8) * (H - 8), "active-mask", "Expected a nonempty, nonrectangular write footprint.");
            Check(sloped > 0 && amber > 0, "anatomy-materials", "Missing sloped ribs or side amber.");
            Check(p.Center.All(x => x >= 85 && x <= W - 85), "center-envelope", "Center leaves the placement envelope.");
            Check(p.Ground.Skip(4).Take(W - 8).All(y => y >= 14 && y <= 76), "ground-envelope", "Ground leaves the shoulder/lip envelope.");
            Check(p.Ribs.Count >= 3 && p.Ribs.Count <= 9, "rib-count", $"ribs={p.Ribs.Count}");
            Check(p.Teeth.Count >= 25 && p.Teeth.Count <= 150, "tooth-count", $"teeth={p.Teeth.Count}");
            Check(p.Fibers.Count >= 5 && p.Fibers.Count <= 70, "fiber-count", $"Expected 5..70 hanging side fibers, got {p.Fibers.Count}.");
            foreach (var rib in p.Ribs)
            {
                Check(In(rib.Root) && In(rib.Tip) && rib.Root != rib.Tip && rib.Roots != null && rib.Roots.Length == 3 &&
                    rib.Roots.Distinct().Count() == 3 && rib.Roots.All(In), "rib-metadata", $"Malformed rib at {rib.Root}.");
                var connected = new HashSet<P.Point>(); var todo = new Stack<P.Point>(); todo.Push(rib.Root);
                bool hostContact = false;
                while (todo.Count > 0)
                {
                    var at = todo.Pop();
                    if (!In(at) || p.Cells[at.X, at.Y].Key != "rib" || !connected.Add(at)) continue;
                    foreach (var d in Directions)
                    {
                        var next = new P.Point(at.X + d.X, at.Y + d.Y);
                        if (In(next) && p.Cells[next.X, next.Y].Write && p.Cells[next.X, next.Y].Key is not null and not "rib") hostContact = true;
                        todo.Push(next);
                    }
                }
                Check(connected.Contains(rib.Tip) && rib.Roots != null && rib.Roots.All(connected.Contains), "rib-continuity", $"Disconnected tip/branch at {rib.Root}.");
                // An exposed root is intentional. The connected rib component needs
                // host contact; requiring every endpoint to be buried would reject it.
                Check(hostContact, "rib-host", $"Rib at {rib.Root} has no solid host contact.");
            }
            var hazards = new bool[W, H]; var objects = new bool[W, H];
            bool CacheNear(int x, int y) => p.Chests.Any(c => x >= c.X - 3 && x <= c.X + 4 && y >= c.Y - 4 && y <= c.Y + 3);
            foreach (var t in p.Teeth)
            {
                Check(t.Variant >= 0 && t.Variant <= 7, "tooth-variant", $"Invalid variant {t.Variant} at {t.Root}.");
                for (int dy = 0; dy < 4; dy++) for (int dx = 0; dx < 4; dx++)
                {
                    int x = t.Root.X + dx, y = t.Root.Y + dy;
                    Check(In(x, y), "tooth-bounds", $"Tooth outside padding at ({x},{y}).");
                    if (!In(x, y)) continue;
                    var c = p.Cells[x, y];
                    Check(c.Write && c.Key == null && c.Slope == 0 && !objects[x, y], "tooth-footprint", $"Occupied/overlapping tooth at ({x},{y}).");
                    Check(!CacheNear(x, y), "cache-hazard-spacing", $"Tooth encroaches cache clearance at ({x},{y}).");
                    hazards[x, y] = objects[x, y] = true;
                }
                for (int i = 0; i < 4; i++)
                {
                    var at = t.Support(i);
                    Check(In(at) && Solid(p.Cells[at.X, at.Y]) && p.Cells[at.X, at.Y].Key != "rib", "tooth-support", $"Invalid support {at} for {t.Root}.");
                }
            }
            Check(p.Chests.Count == 1 && p.NodeSites.Count == 1 && p.Pockets.Count == 3, "pocket-counts", "Expected one cache chest, one node site and three pockets.");
            foreach (string role in new[] { "cache", "exploration", "node-reservation" })
                Check(p.Pockets.Count(v => v.Role == role) == 1, "pocket-role", "Expected exactly one " + role + " pocket.");
            foreach (var pocket in p.Pockets)
            {
                Check(pocket.RadiusX > 0 && pocket.RadiusY > 0 && In(pocket.Center.X - pocket.RadiusX, pocket.Center.Y - pocket.RadiusY) &&
                    In(pocket.Center.X + pocket.RadiusX, pocket.Center.Y + pocket.RadiusY), "pocket-envelope", $"Malformed {pocket.Role} envelope.");
                Check(In(pocket.Center) && p.Cells[pocket.Center.X, pocket.Center.Y].Write && p.Cells[pocket.Center.X, pocket.Center.Y].Side,
                    "pocket-center", $"Missing side pocket at {pocket.Center}.");
            }
            var cache = p.Pockets.First(v => v.Role == "cache");
            var exploration = p.Pockets.First(v => v.Role == "exploration");
            var node = p.Pockets.First(v => v.Role == "node-reservation");
            int gap = exploration.Center.Y - exploration.RadiusY - (cache.Center.Y + cache.RadiusY);
            Check(gap >= 50, "cave-spacing", $"Independent cave envelope gap is {gap}, expected >=50.");
            Check(node.RadiusX < exploration.RadiusX && node.RadiusY < exploration.RadiusY, "node-size", "Node reservation must be smaller than its exploration cave.");
            foreach (var c in p.Chests)
            {
                Check(Math.Abs(c.X - cache.Center.X) < cache.RadiusX && Math.Abs(c.Y - cache.Center.Y) < cache.RadiusY,
                    "chest-role", $"Chest {c} is outside the cache envelope.");
                for (int dx = 0; dx < 2; dx++)
                {
                    for (int dy = 0; dy < 2; dy++)
                    {
                        int x = c.X + dx, y = c.Y + dy;
                        Check(In(x, y) && p.Cells[x, y].Write && p.Cells[x, y].Key == null && p.Cells[x, y].Side && !objects[x, y],
                            "chest-footprint", $"Invalid chest footprint at ({x},{y}).");
                        if (In(x, y)) objects[x, y] = true;
                    }
                    Check(In(c.X + dx, c.Y + 2) && Solid(p.Cells[c.X + dx, c.Y + 2]), "chest-support", $"Missing chest foot for {c}.");
                }
            }
            foreach (var at in p.NodeSites) Check(at == node.Center, "node-role", $"Node site {at} does not match its reserved pocket.");
            foreach (var f in p.Fibers)
            {
                Check(In(f.Root) && p.Cells[f.Root.X, f.Root.Y].Write && p.Cells[f.Root.X, f.Root.Y].Key == "grass" &&
                    p.Cells[f.Root.X, f.Root.Y].Slope == 0 && f.Length >= 2 && f.Length <= 6, "fiber-anchor", $"Invalid strand at {f.Root}, length={f.Length}.");
                for (int d = 1; d <= f.Length; d++)
                {
                    int x = f.Root.X, y = f.Root.Y + d;
                    Check(In(x, y) && p.Cells[x, y].Write && p.Cells[x, y].Key == null && !objects[x, y] && !CacheNear(x, y),
                        "fiber-footprint", $"Blocked/overlapping strand at ({x},{y}).");
                    if (In(x, y)) objects[x, y] = true;
                }
                Check(In(f.Root.X, f.Root.Y + 1) && p.Cells[f.Root.X, f.Root.Y + 1].Side, "fiber-side", "Strand is not in a side passage.");
            }
            bool[,] expected = Flood(p, hazards), actual = p.Reachable();
            bool equal = true;
            for (int y = 0; y < H; y++) for (int x = 0; x < W; x++) equal &= expected[x, y] == actual[x, y];
            Check(equal, "reach-oracle", "Reachable disagrees with the independent 2x3 occupancy flood including all tooth rectangles.");
            void Reached(P.Point at, string name) => Check(In(at) && expected[at.X, at.Y], "reach/" + name, $"Unreachable {name} at {at} with teeth present.");
            Reached(p.Entrance, "entrance"); Reached(p.Exit, "outlet");
            foreach (var c in p.Chests) Reached(new(c.X, c.Y - 2), "cache");
            foreach (var at in p.NodeSites) Reached(new(at.X, at.Y - 1), "node");
            r.Range("writtenCells", active); r.Range("slopedCells", sloped); r.Range("ribs", p.Ribs.Count);
            r.Range("teeth", p.Teeth.Count); r.Range("fibers", p.Fibers.Count); r.Range("caveEnvelopeGap", gap);
        }
        // Build all legal top-left body positions before flooding a flat queue.
        // Sloped solids conservatively block their full cell. Fibers are decoration.
        private static bool[,] Flood(P p, bool[,] hazards)
        {
            var fits = new bool[W * H]; var seen = new bool[W, H]; var queue = new int[W * H];
            for (int y = 4; y + 2 < H - 4; y++) for (int x = 4; x + 1 < W - 4; x++)
            {
                bool free = true;
                for (int dy = 0; dy < 3; dy++) for (int dx = 0; dx < 2; dx++)
                { var c = p.Cells[x + dx, y + dy]; free &= c.Write && c.Key == null && !hazards[x + dx, y + dy]; }
                fits[y * W + x] = free;
            }
            int head = 0, tail = 0;
            if (In(p.Entrance) && fits[p.Entrance.Y * W + p.Entrance.X])
            { queue[tail++] = p.Entrance.Y * W + p.Entrance.X; seen[p.Entrance.X, p.Entrance.Y] = true; }
            while (head < tail)
            {
                int at = queue[head++];
                foreach (int next in new[] { at - 1, at + 1, at - W, at + W })
                    if (next >= 0 && next < fits.Length && fits[next] && !seen[next % W, next / W])
                    { seen[next % W, next / W] = true; queue[tail++] = next; }
            }
            return seen;
        }
        private static void InvalidInputs(Report r)
        {
            var cases = new (string name, Action action)[] {
                ("short-spine", () => P.Create(42, new int[H - 1])),
                ("long-spine", () => P.Create(42, new int[H + 1])),
                ("empty-spine", () => P.Create(42, Array.Empty<int>())),
                ("left-shoulder-low", () => P.Create(42, null, 39, 56)),
                ("left-shoulder-high", () => P.Create(42, null, 77, 56)),
                ("right-shoulder-low", () => P.Create(42, null, 56, 39)),
                ("right-shoulder-high", () => P.Create(42, null, 56, 77)),
                ("spine-outside-left", () => P.Create(42, Enumerable.Repeat(0, H).ToArray())),
                ("spine-outside-right", () => P.Create(42, Enumerable.Repeat(W, H).ToArray()))
            };
            foreach (var test in cases)
            {
                r.InvalidInputCases++;
                try { test.action(); r.AddFailure("input/" + test.name, "Create", "Invalid input was accepted."); }
                catch (ArgumentException) { r.RejectedInputs++; }
                catch (Exception e) { r.AddFailure("input/" + test.name, "Create", "Uncontrolled exception: " + e.GetType().Name); }
            }
        }
        private static P.Point Find(P p, Func<P.Cell, bool> predicate)
        {
            for (int y = 4; y < H - 4; y++) for (int x = 4; x < W - 4; x++) if (predicate(p.Cells[x, y])) return new(x, y);
            throw new InvalidOperationException("Mutation fixture cell not found.");
        }
        private static void Mutations(Report r)
        {
            var tests = new (string name, Action<P> corrupt)[] {
                ("missing-rib", p => { var at = p.Ribs[0].Root; p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Key = null, Slope = 0 }; }),
                ("wall-fringe", p => { int x = p.Entrance.X; int y = p.Ground[x] + 1; p.Cells[x, y] = p.Cells[x, y] with { Wall = "stone", Write = true }; }),
                ("blocked-outlet", p => { for (int dx = 0; dx < 2; dx++) for (int dy = 0; dy < 3; dy++) { int x = p.Exit.X + dx, y = p.Exit.Y + dy; p.Cells[x, y] = p.Cells[x, y] with { Key = "stone" }; } }),
                ("missing-tooth-support", p => { var at = p.Teeth[0].Support(0); p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Key = null }; }),
                ("extra-chest-role", p => { int i = p.Pockets.FindIndex(v => v.Role == "exploration"); p.Pockets[i] = p.Pockets[i] with { Role = "cache" }; }),
                ("extra-chest-object", p => p.Chests.Add(p.Chests[0])),
                ("side-amber-false", p => { var at = Find(p, c => c.Key == "amber"); p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Side = false }; }),
                ("side-amber-wall-false", p => { var at = Find(p, c => c.Wall == "amber" && c.Key == null); p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Side = false }; }),
                ("malformed/tooth-variant", p => p.Teeth[0] = p.Teeth[0] with { Variant = 8 }),
                ("malformed/tooth-negative-variant", p => p.Teeth[0] = p.Teeth[0] with { Variant = -4 }),
                ("malformed/tooth-bounds", p => p.Teeth[0] = p.Teeth[0] with { Root = new(-10, 60) }),
                ("malformed/rib-null-roots", p => p.Ribs[0] = p.Ribs[0] with { Roots = null }),
                ("malformed/rib-empty-roots", p => p.Ribs[0] = p.Ribs[0] with { Roots = Array.Empty<P.Point>() }),
                ("malformed/rib-duplicate-roots", p => p.Ribs[0] = p.Ribs[0] with { Roots = Enumerable.Repeat(p.Ribs[0].Roots[0], 3).ToArray() }),
                ("malformed/pocket-radius", p => p.Pockets[0] = p.Pockets[0] with { RadiusX = 0 }),
                ("malformed/pocket-bounds", p => p.Pockets[0] = p.Pockets[0] with { Center = new(-10, p.Pockets[0].Center.Y) }),
                ("malformed/pocket-role", p => p.Pockets[0] = p.Pockets[0] with { Role = "unknown" }),
                ("malformed/node-bounds", p => p.NodeSites[0] = new(-1, 100)),
                ("malformed/center", p => p.Center[H / 2] = -1),
                ("malformed/ground", p => p.Ground[4] = -1),
                ("malformed/fiber-length", p => p.Fibers[0] = p.Fibers[0] with { Length = 0 }),
                ("missing-fiber-anchor", p => { var at = p.Fibers[0].Root; p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Key = null }; }),
                ("overlapping-teeth", p => p.Teeth[1] = p.Teeth[0]),
                ("blocked-tooth-footprint", p => { var at = p.Teeth[0].Root; p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Key = "stone" }; }),
                ("rib-tooth-support", p => { var at = p.Teeth[0].Support(0); p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Key = "rib" }; }),
                ("missing-chest-support", p => { var at = p.Chests[0]; p.Cells[at.X, at.Y + 2] = p.Cells[at.X, at.Y + 2] with { Key = null }; }),
                ("insufficient-cave-spacing", p => p.Pockets[1] = p.Pockets[1] with { Center = new(p.Pockets[1].Center.X, p.Pockets[0].Center.Y + 20) }),
                ("outside-write-padding", p => p.Cells[0, 0] = new("stone", null, 0, false, true)),
                ("inactive-cell-data", p => { var at = Find(p, c => c.Write && c.Key == "stone"); p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Write = false }; }),
                ("unknown-wall", p => { var at = Find(p, c => c.Wall == "stone"); p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Wall = "unknown" }; }),
                ("invalid-slope", p => { var at = Find(p, c => c.Key == "rib"); p.Cells[at.X, at.Y] = p.Cells[at.X, at.Y] with { Slope = 5 }; })
            };
            foreach (var test in tests)
            {
                r.MutationCases++;
                var control = new Control { Name = test.name }; r.Controls.Add(control);
                P p;
                try { p = P.Create(42); p.Validate(); test.corrupt(p); }
                catch (Exception e)
                {
                    control.Status = "SETUP-FAILED"; control.Detail = e.GetType().Name + ": " + e.Message;
                    r.AddFailure("mutation/" + test.name, "seed=42", control.Detail); continue;
                }
                // Only the real validator decides whether this mutation is rejected.
                // Setup errors and incidental index/null crashes are not a pass.
                try { p.Validate(); control.Status = "ACCEPTED"; control.Detail = "Corrupt plan survived the real Validate()."; }
                catch (InvalidOperationException e) { control.Status = "REJECTED"; control.Detail = e.Message; r.RejectedMutations++; }
                catch (Exception e) { control.Status = "UNCONTROLLED-EXCEPTION"; control.Detail = e.GetType().Name + ": " + e.Message; }
                if (control.Status != "REJECTED") r.AddFailure("mutation/" + test.name, "seed=42", control.Status + ": " + control.Detail);
            }
        }
        public static string Svg(int seed, bool suitePassed, string sourceHash)
        {
            P p = P.Create(seed);
            string Escape(string value) => System.Security.SecurityElement.Escape(value) ?? "";
            var s = new StringBuilder();
            s.Append($"<svg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 {W} {H}' width='{W}' height='{H}' role='img'>");
            s.Append($"<title>Maw seed {seed}: native tile-coordinate geometry</title><desc>Suite {(suitePassed ? "PASS" : "FAIL")}; source SHA256 {Escape(sourceHash)}. Diagram only; no game render or movement proof. Yellow rectangles are complete hazardous tooth footprints; orange lines are hanging fiber. Transparent space is outside the write mask.</desc>");
            s.Append("<g id='active-mask' fill='#20252b'>");
            for (int y = 0; y < H; y++) for (int x = 0; x < W; x++) if (p.Cells[x, y].Write)
                s.Append($"<rect x='{x}' y='{y}' width='1' height='1'/>");
            s.Append("</g><g id='native-cells'>");
            for (int y = 0; y < H; y++) for (int x = 0; x < W; x++)
            {
                var c = p.Cells[x, y]; if (!c.Write) continue;
                if (c.Wall != null)
                {
                    string wall = c.Wall switch { "soil" => "#372923", "grass" => "#49351d", "amber" => "#8a5514", _ => "#303034" };
                    s.Append($"<rect x='{x}' y='{y}' width='1' height='1' fill='{wall}'/>");
                }
                if (c.Key == null) continue;
                string fill = c.Key switch { "cap" => "#c99a48", "grass" => "#a97932", "soil" => "#735345", "stone" => "#514b49", "rib" => "#e8dfc4", "amber" => "#efa71c", _ => "#ff00ff" };
                // Native slope occupancy: 1 y>=x, 2 y>=1-x,
                // 3 y<=1-x, 4 y<=x (see Test-MawRibContour.ps1).
                string points = c.Slope switch {
                    1 => $"{x},{y} {x},{y + 1} {x + 1},{y + 1}",
                    2 => $"{x + 1},{y} {x},{y + 1} {x + 1},{y + 1}",
                    3 => $"{x},{y} {x + 1},{y} {x},{y + 1}",
                    4 => $"{x},{y} {x + 1},{y} {x + 1},{y + 1}", _ => null
                };
                if (points == null) s.Append($"<rect x='{x}' y='{y}' width='1' height='1' fill='{fill}'/>");
                else s.Append($"<polygon points='{points}' fill='{fill}' data-slope='{c.Slope}'/>");
            }
            s.Append("</g><g id='teeth' fill='#ffe600' fill-opacity='.8'>");
            foreach (var t in p.Teeth) s.Append($"<rect x='{t.Root.X}' y='{t.Root.Y}' width='4' height='4' data-variant='{t.Variant}'><title>Tooth face {t.Face}, variant {t.Variant}</title></rect>");
            s.Append("</g><g id='hanging-fiber' stroke='#ed932c' stroke-width='.55'>");
            foreach (var f in p.Fibers) s.Append(FormattableString.Invariant($"<path d='M {f.Root.X + .5} {f.Root.Y + 1} v {f.Length}'><title>Hanging fiber, {f.Length} cells</title></path>"));
            s.Append("</g><g id='chests' fill='#e35a4a'>");
            foreach (var c in p.Chests) s.Append($"<rect x='{c.X}' y='{c.Y}' width='2' height='2'/>");
            s.Append("</g><g id='pocket-labels' font-family='sans-serif' font-size='2.8' fill='#ffffff' stroke='#13171c' stroke-width='.18' paint-order='stroke'>");
            foreach (var pocket in p.Pockets)
                s.Append($"<text x='{pocket.Center.X}' y='{pocket.Center.Y}' text-anchor='middle'>{Escape(pocket.Role)}</text>");
            s.Append("</g><g id='route-targets' fill='none' stroke='#38dfed' stroke-width='.4'>");
            foreach (var at in new[] { p.Entrance, p.Exit }) s.Append($"<rect x='{at.X}' y='{at.Y}' width='2' height='3'/>");
            foreach (var at in p.NodeSites) s.Append($"<circle cx='{at.X}' cy='{at.Y}' r='2'/>");
            s.Append("</g></svg>"); return s.ToString();
        }
